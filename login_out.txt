            (function () {
                var logoutForm = document.getElementById('smsLogoutForm');
                if (!logoutForm) {
                    return;
                }

                logoutForm.addEventListener('submit', function (event) {
                    event.preventDefault();

                    if (window.smsClearAuthTokens) {
                        window.smsClearAuthTokens();
                    }

                    fetch('/logout', {
                        method: 'POST',
                        headers: {
                            Accept: 'text/html'
                        },
                        credentials: 'same-origin'
                    }).finally(function () {
                        window.location.replace('/login?logout');
                    });
                });
            })();
        </script>
